cmake_minimum_required(VERSION 3.20)
project(gem_launcher LANGUAGES CXX RC)

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Gates of Hell is 64-bit: configure with -A x64")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(EXISTS ${CMAKE_SOURCE_DIR}/launcher/res/title.ttf)
    file(WRITE ${CMAKE_SOURCE_DIR}/launcher/fonts.rc "400 RCDATA \"res\\\\title.ttf\"\n")
else()
    file(WRITE ${CMAKE_SOURCE_DIR}/launcher/fonts.rc "// no title.ttf - title falls back to Impact\n")
endif()

add_executable(gem_launcher WIN32 launcher/main.cpp launcher/launcher.rc)
set_target_properties(gem_launcher PROPERTIES
    OUTPUT_NAME "GEM Launcher"
    LINK_FLAGS "/MANIFEST:NO")   # manifest comes from launcher.rc
target_link_libraries(gem_launcher PRIVATE shlwapi psapi shell32 advapi32 user32 gdi32 msimg32 comctl32 winmm bcrypt winhttp)
# NOTE: payload.rc / payload.manifest are generated by build_zig.bat; run it once before a CMake build.

add_library(gem_ext SHARED gem_ext_stub/dllmain.cpp)
target_link_libraries(gem_ext PRIVATE psapi)

set_target_properties(gem_launcher gem_ext PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY         ${CMAKE_BINARY_DIR}/dist
    RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/dist
    RUNTIME_OUTPUT_DIRECTORY_DEBUG   ${CMAKE_BINARY_DIR}/dist)

configure_file(${CMAKE_SOURCE_DIR}/gem_launcher.ini ${CMAKE_BINARY_DIR}/dist/gem_launcher.ini COPYONLY)
